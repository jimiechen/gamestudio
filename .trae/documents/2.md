import hashlib
import hmac
import json
import time
from typing import Dict, Any


def generate_signature(timestamp: str, params: Dict[str, Any], secret_key: str) -> str:
"""
生成 HMAC-SHA256 签名

:param timestamp: 时间戳（毫秒）
:param params: 请求参数
:param secret_key: 密钥
:return: 签名字符串
"""
# 将字典转换为紧凑格式的JSON字符串，并确保浮点数1.0保持为1.0
json_str = json.dumps(
params,
separators=(',', ':'),
ensure_ascii=False  # 保持中文不转义
)
string_to_sign = timestamp + "="
if params:
string_to_sign += json_str
print("待签名字符串:", string_to_sign)
# 生成HMAC-SHA256签名
signature = hmac.new(
secret_key.encode('utf-8'),
string_to_sign.encode('utf-8'),
hashlib.sha256
).hexdigest()

print("生成验签:", signature)
return signature
