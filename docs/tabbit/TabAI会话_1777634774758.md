```gdscript
const TEST_DIRS: Array[String] = [
	"res://tests/unit",
	"res://tests/integration",
	"res://tests/ui",
	"res://tests/e2e"
]

func _run_all_tests() -> void:
	var test_files: Array[String] = _discover_test_files()

	test_files.sort()

	for test_file: String in test_files:
		_run_test_file(test_file)

func _discover_test_files() -> Array[String]:
	var result: Array[String] = []

	for test_dir: String in TEST_DIRS:
		_collect_test_files(test_dir, result)

	return result

func _collect_test_files(dir_path: String, result: Array[String]) -> void:
	var dir: DirAccess = DirAccess.open(dir_path)

	if dir == null:
		push_warning("测试目录不存在: %s" % dir_path)
		return

	dir.list_dir_begin()

	while true:
		var file_name: String = dir.get_next()

		if file_name == "":
			break

		if file_name.begins_with("."):
			continue

		var full_path: String = dir_path.path_join(file_name)

		if dir.current_is_dir():
			_collect_test_files(full_path, result)
		elif file_name.begins_with("test_") and file_name.ends_with(".gd"):
			result.append(full_path)

	dir.list_dir_end()
```
